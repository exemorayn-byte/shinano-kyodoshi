<script>
let database = [];

// CSV読み込み
Papa.parse("data.csv", {
    download: true,
    header: true,
    complete: function(results) {
        database = results.data.filter(row => row.title);
        displayResults(database);
        document.getElementById('count').innerText = `${database.length} 件のデータが見つかりました`;
    }
});

// 検索実行
document.getElementById('search-box').addEventListener('input', function(e) {
    const query = e.target.value.toLowerCase();
    const filtered = database.filter(row => {
        return row.title.toLowerCase().includes(query) || 
               row.author.toLowerCase().includes(query) || 
               row.note.toLowerCase().includes(query);
    });
    displayResults(filtered);
    document.getElementById('count').innerText = `${filtered.length} 件ヒットしました`;
});

function displayResults(data) {
    const container = document.getElementById('results');
    container.innerHTML = '';
    
    data.forEach(row => {
        const card = document.createElement('div');
        card.className = 'card';
        card.innerHTML = `
            <span class="mag-tag">${row.magazine}</span>
            <span class="title">${row.title}</span>
            <div class="author">${row.author || '不明'}</div>
            <div class="meta">
                ${row.date.split('/')[0]}年 / ${row.volume}巻 ${row.issue}号 (通巻${row.serial_num}号) / ${row.page}ページ
            </div>
            ${row.note ? `<div class="note">備考: ${row.note}</div>` : ''}
        `;
        container.appendChild(card);
    });
}
</script>
